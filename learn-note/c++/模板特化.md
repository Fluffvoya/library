# 特化
## 全特化
```cpp
#include <iostream>
template<typename T_>
void func(T_){
	std::cout<<"wide";
}

template<>
void func<int>(int){
	std::cout<<"int";
}

int main(){
	func(1.1);
	func(1);
}
```
## 偏特化
```cpp
#include <iostream>

template<typename T_>
constexpr int v = 6;
template<typename T_>
constexpr int v<const T_> = 5;

int main() {
	std::cout<<v<int><<std::endl;
	std::cout<<v<const int><<std::endl;
}
```
或者
```cpp
template<typename T_,typename U_>
void func(T_,U_){
	std::cout<<"wide";
}

template<typename T_>
void func<T_,int>(T_,int){
	std::cout<<"int";
}
```
> 偏特化不能出现在函数模板里
